% 3 · el bucle ReAct con un rechazo real ---------------------------------------------
\begin{frame}{El bucle ReAct y un rechazo real}
  \centering
  \begin{tikzpicture}[x=1mm, y=1mm, every node/.append style={anchor=north west}]
    \node[traza, text width=37mm] (q) at (0,0) {%
      \textsf{\footnotesize\bfseries MBPP 11}\\[1mm]
      Write a python function to remove first\\ and last occurrence of a given character.\\[1mm]
      assert remove\_Occ("hello","l") == "heo"\\
      assert remove\_Occ("abcda","a") == "bcd"\\
      assert remove\_Occ("PHP","P") == "H"};
    \node[traza, text width=37mm, draw=cOurs!60, fill=cOurs!5] (r1) at (0,-31) {%
      \textsf{\footnotesize\bfseries Ronda 1 \textperiodcentered\ \no\ rechazada}\\[1mm]
      \textsf{\scriptsize El Auditor \textbf{ejecuta} el código con las pruebas:}\\[0.8mm]
      assert remove\_Occ("abcda","a") == "bcd"\\
      {\color{cOurs}AssertionError}};
    \node[traza, text width=80mm] (r2) at (45,0) {%
      \textsf{\footnotesize\bfseries Ronda 2 \textperiodcentered\ el error vuelve al Investigador}\\[1mm]
      \kw{Thought:} The previous solution did not work because it\\ used `replace' \dots{} We need to find the indices of the\\ first and last occurrences \dots\\
      \kw{Action:} python[def remove\_Occ(s, ch):\\
      \ \ \ \ first\_index = s.find(ch)\\
      \ \ \ \ last\_index = s.rfind(ch)\\
      \ \ \ \ \dots\\
      \ \ \ \ return s[:first\_index] + s[first\_index+1:last\_index]\\
      \ \ \ \ \ \ \ \ \ \ \ + s[last\_index+1:]\\[0.6mm]
      assert remove\_Occ("abcda", "a") == "bcd" \dots\\
      print("ok")]\\
      \kw{Observation:} \obs{ok}\\
      \kw{Thought:} The code passed all the tests \dots\\
      \kw{Final:} \textsf{el mismo código}};
    \node[traza, text width=80mm, draw=cOk!70, fill=cOk!6] (ok) at ([yshift=-2mm]r2.south west) {%
      \textsf{\footnotesize\bfseries Auditor \textperiodcentered\ \si\ aprobada:} \textsf{\scriptsize las 3 pruebas pasan; el Redactor entrega el código sin reescribirlo}};
    \draw[ar, draw=cOurs!70] (r1.east) -- ++(4,0) |- ([yshift=-8mm]r2.north west);
  \end{tikzpicture}\\[1mm]
  {\scriptsize\color{cSrc}Traza real de la corrida, abreviada en «…». El modelo escribe Thought y Action; la Observation la escribe el sistema.\par}
\end{frame}
